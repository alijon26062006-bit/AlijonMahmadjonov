Открыть документацию 
import uuid, requests

r = requests.post(
    "{{ request.base_url }}api/v1/orders",
    headers={"X-API-Key": KEY,
             "Idempotency-Key": str(uuid.uuid4())},
    json={"product_id": "tg-stars", "quantity": 100,
          "fields": {"telegram_username": "@player"}},
)
print(r.json()["order"]["status"])
